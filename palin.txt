<?php
// PHP code to check for Palindrome string in PHP Using strrev()
$pst = $_POST['pal'];
function Palindrome($string)
{  
	if (strrev($string) == $string)
	{  
	return 1;  
	}
	else
	{
	return 0;
	}
}  
	// Driver Code

	echo "<center><b>Palindrome </b></center>";  
}

{

}
