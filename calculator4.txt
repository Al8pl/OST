<body>
<h3>Simple Calculator</h3>

<form method="post">
    Enter First Number:
    <input type="number" name="a" step="any" required><br><br>

    Enter Second Number:
    <input type="number" name="b" step="any" required><br><br>

    Enter Operator:
    <input type="text" name="op" required><br><br>

    <input type="submit" name="btn" value="Calculate">
</form>

<?php
if (isset($_POST['btn'])) {
    $a = $_POST['a'];
    $b = $_POST['b'];
    $op = $_POST['op'];

    switch ($op) {
        case "+":
            $r = $a + $b;
            break;

        case "-":
            $r = $a - $b;
            break;

        case "*":
            $r = $a * $b;
            break;

        case "/":
            if ($b == 0) 
                $r = "Cannot divide by zero";
            else
                $r = $a / $b;
            break;

        default:
            $r = "Invalid operator";
    }

    echo "<h3>Result: $r</h3>";
}
?>
